{

    while( *pLista && ( cmp = cmp((*pLista)->info, dato)) > 0 )
       pLista = &(*pLista)->sig;

    if(*pLista &&  cmp==0){

    }

    nue = (tNodo*)malloc(sizeof(tNodo));

    if(!nue || !(nue->info = malloc(tamDato))){
